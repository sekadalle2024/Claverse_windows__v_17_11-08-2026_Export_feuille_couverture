// 5. Simuler le figement manuellement
const numCols = 3;
const rows = table.querySelectorAll('tr');
rows.forEach(row => {
  const cells = row.querySelectorAll('th, td');
  let left = 0;
  cells.forEach((cell, idx) => {
    if (idx < numCols) {
      cell.style.position = 'sticky';
      cell.style.left = `${left}px`;
      cell.style.zIndex = '100';
      cell.style.backgroundColor = '#ffffff';
      console.log(`Cell ${idx}: left=${left}px`);
    }
    left += cell.offsetWidth;
  });
});

// 6. Vérifier que le style a bien été appliqué

// ATTENTION: À utiliser uniquement pour tester sans modifier menu.js

// 1. Définir la fonction ensureScrollContainer
window.ensureScrollContainer = function(table) {
  const parent = table.parentElement;
  if (parent.classList.contains('freeze-scroll-container')) return;
  parent.classList.add('freeze-scroll-container');
  parent.style.overflowX = 'auto';
  parent.style.overflowY = 'visible';
  parent.style.maxWidth = '100%';
  parent.style.position = 'relative';
  console.log('✓ Conteneur configuré');
};

// 2. Définir la fonction applyFreezePanes corrigée
window.applyFreezePanes = function(table, freezeRows, freezeCols) {
  console.log(`🔧 Figement: ${freezeCols} colonnes`);
  table.style.borderCollapse = 'separate';
  table.style.borderSpacing = '0';
  table.style.minWidth = 'max-content';
  
  void table.offsetHeight; // Force reflow
  
  const rows = Array.from(table.querySelectorAll('tr'));
  rows.forEach((row, rowIndex) => {
    const cells = Array.from(row.querySelectorAll('th, td'));
    let currentLeft = 0;
    cells.forEach((cell, colIndex) => {
      if (colIndex < freezeCols) {
        cell.style.position = 'sticky';
        cell.style.left = `${currentLeft}px`;
        cell.style.zIndex = '100';
        cell.style.backgroundColor = cell.tagName === 'TH' ? '#380101' : 
          (rowIndex % 2 === 0 ? '#ffffff' : '#f9fafb');
        if (colIndex === freezeCols - 1) {
          cell.style.borderRight = '3px solid #380101';
          cell.style.boxShadow = '2px 0 5px rgba(56,1,1,0.2)';
        }
      } else {
        cell.style.position = '';
        cell.style.left = '';
        cell.style.zIndex = '';
      }
      currentLeft += cell.offsetWidth || 150;
    });
  });
  console.log('✅ Figement terminé');
};

// 3. Tester immédiatement
const table = document.querySelector('table');
ensureScrollContainer(table);
applyFreezePanes(table, 0, 3);
